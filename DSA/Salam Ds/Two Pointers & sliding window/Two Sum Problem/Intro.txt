Brute Force approach: O(n^2)
var twoSum = function(arr, target) {
  let output=[];
for (let i=0;i<arr.length;i++){
    for (let j=i+1;j<arr.length;j++){
          if (arr[i]+arr[j]==target){
           output.push(i,j)
          }
    }
}
return output;
};

In Question,it is mentioned that we dont have to solve it with O(n^2).

So ,recommended solution is
var twoSum = function(arr, target) {
 let map={},output=[];
for (let i=0;i<arr.length;i++){
    map[arr[i]]=i
}
for (let i=0;i<arr.length;i++){
    let found=target-arr[i]
    if (map[found] && map[found] !=i ){// map[found] !=i  helps to prevent duplication
       return[i,map[found]]
       
    }
}
};
