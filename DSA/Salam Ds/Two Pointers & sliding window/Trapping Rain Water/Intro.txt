var trap = function(arr) {
   let n=arr.length,maxL=[],maxR=[];
   //step 1 : fill the maxL array
    maxL[0]=arr[0];
   for (let i=1;i<n;i++ ){
    maxL[i]=Math.max(maxL[i-1],arr[i])
   }
   //step 2 : fill the maxR array
    maxR[n-1]=arr[n-1];
   for (let i=n-2;i>=0;--i){
    maxR[i]=Math.max(maxR[i+1],arr[i])
   }

   //step 3: find the ans 
   let ans=0;
   for (let i=0;i<n;i++){
    let waterTapped=(Math.min(maxL[i],maxR[i])-arr[i])
    ans=ans+(waterTapped>0?waterTapped:0)
   }
   return ans
};
