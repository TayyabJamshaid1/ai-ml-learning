Whenever you have to find a substring in your question,allways think about a sliding window.
First i have to use a sliding window.Then i have to check that map obj is valid or not.So our program contain 2 algorithms/stages

a)Sliding a window
b)Check if map is valid

var characterReplacement = function(s, k) {
    let i=j=0,map={},maxWindow=0;
    map[s[i]]=1;
    while(j<s.length){
        if (isWindowValid(map,k)){
      maxWindow=Math.max(maxWindow,j-i+1)
      ++j;
      map[s[j]]=(map[s[j]]||0)+1
        }else{
            --map[s[i]];
            ++i
        }
    }
    return maxWindow
};
var isWindowValid = function(map, k) {
let maxCount=totalSum=0;
for (let i=0;i<26;i++){
    let alphabet=String.fromCharCode(i+65);
    let mapped=map[alphabet]!=undefined?map[alphabet]:0
    totalSum=totalSum+mapped
    maxCount=Math.max(maxCount,mapped)
}

let res=totalSum-maxCount<=k;
return res
}

Time complexity is O(n),space complexity is O(1)
