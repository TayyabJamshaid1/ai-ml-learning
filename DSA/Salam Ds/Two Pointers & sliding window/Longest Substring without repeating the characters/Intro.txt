Whenever you have to find a substring in your question,allways think about a sliding window

/**
 * @param {string} s
 * @return {number}
 */
var lengthOfLongestSubstring = function(s) {
    let i=j=0,map={},max=0
    while(j<s.length){
        //check that if character is already avaiable in map and it is on right side,means duplicated character index obviously  greater then i,verify it also
        if (map[s[j]] != undefined && map[s[j]]>=i){ // >= is used because if first character is repeated ,its value is 0,so thats why we use = 
            i=map[s[j]]+1
         } 

        map[s[j]]=j
        let currentIterationLength=j-i+1 //max string calculation formula
        max=currentIterationLength>max?currentIterationLength:max
        ++j;
    }
    return max
};
