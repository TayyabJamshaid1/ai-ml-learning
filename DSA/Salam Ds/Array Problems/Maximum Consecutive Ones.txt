let arr=[1,1,0,1,1,1],x=0,number_of_consecutives=0;
function getConsecutiveNumber(arr){
for (let i=0;i<arr.length;i++){
  if (arr[i]!=0){
    x++
   
  }else{
    number_of_consecutives=Math.max(x,number_of_consecutives)
    x=0
  }
}
return Math.max(x,number_of_consecutives)
}
