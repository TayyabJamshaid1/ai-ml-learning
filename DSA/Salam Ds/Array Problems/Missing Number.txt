let arr=[9,6,4,2,3,5,7,0,1],missingNumber,sumOfArray=0;  
function getMissingNumber(arrSend){
  let n=arrSend.length;
  let totalSum=(n*(n+1))/2

        for (let i=0;i<n;i++){
           sumOfArray+=arrSend[i]
        }
        missingNumber=totalSum-sumOfArray
        return missingNumber
}
