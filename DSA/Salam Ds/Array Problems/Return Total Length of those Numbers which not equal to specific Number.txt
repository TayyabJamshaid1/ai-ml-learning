The Golden Rule to Remember
Whenever you use splice(), shift(), or unshift() inside a loop, your algorithm becomes O(n²)!

function removeSpecificNumber(arr,num){
    let x=0;
    for (let i=0;i<arr.length;i++){
        if (arr[i]!=num){
           arr[x]=arr[i]
           x+=1
        }
    }
    return {arr,x}
}
