let arr=[3,1,5,9,1,5,3],obj={},achiever=0


for (let i=0;i<arr.length;i++){
  obj[arr[i]]=(obj[arr[i]]||0)+1

}

for (let i=0;i<arr.length;i++){
  if(obj[arr[i]]==1){
    achiever=arr[i]
    break
  }
}
console.log(achiever)


Its Time complexity is O(n) and space complexity is also o(n)
