                       arr3.push(arr2[k])

                    }

var nextGreaterElement = function(subsetArray, arr) {
    let res=[], stack = [], nge = {};
    for (let i = arr.length - 1; i >= 0; i--) {
    if (i == arr.length - 1) {
        nge[arr[i]] = -1;
        stack.push(arr[i])
    } else {
        let top = stack[stack.length - 1]
        if (arr[i] < top) {
            nge[arr[i]] = top;
            stack.push(arr[i])

        } else if (arr[i] > top) {
            let found = false;
            let n = stack.length
            for (let j = 0; j < n; j++) {

                let popElement = stack.pop()

                if (popElement && arr[i] < popElement) {

                    found = true;
                    nge[arr[i]] = popElement;
                    stack.push(popElement)

                      stack.push(arr[i])
                    break;
                }
            }
            if (!found) {
                nge[arr[i]] = -1;
                   stack.push(arr[i])

            }

        }
    }
}
for (let i=0;i<subsetArray.length;i++){

   res.push(nge[subsetArray[i]])
    }
}
