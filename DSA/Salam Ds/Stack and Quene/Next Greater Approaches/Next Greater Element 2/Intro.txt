/**
 * @param {number[]} nums
 * @return {number[]}
 */
var nextGreaterElements = function(arr) {
   arr=arr.concat(...arr)
  let n=arr.length
  let res=Array(n).fill(-1),stack = [];
  stack.push(n-1)
  for (let i=n-2;i>=0;--i){
    while(stack.length){
        let top=stack[stack.length-1]
        if (arr[i]<arr[top]){
            res[i]=arr[top]
            break
        }else{
stack.pop()
        }
    }
    if (!stack.length){ //not manadatory step because by default we set all the values with -1 in res array
    res[i]=-1

    }
    stack.push(i)
  }
 
return res.slice(0,n/2);
}

stack.pop()

return res.slice(0,n/2);
}

 var nextGreaterElements = function(arr) {
   arr=arr.concat(...arr)
  let n=arr.length;
  let res=Array(n).fill(-1),stack = [];
  stack.push(arr[n-1])
  for (let i=(2*n)-2;i>=0;--i){   //i take i=(2*n)-2 because if n=9,if i double the n,it will become 18,now i have to start from 16 because 17 is the last index ,we gave to step back from last index
    while(stack.length){
        let top=stack[stack.length-1]
        if (arr[i%n]<top){           //let i =15 ,i take i%n means 15%9 =6 and after some time,i=6,so 6%9=6 so it means 15 and 6 both are pointing same number ,see image(Recommended Approach dry run)
            res[i%n]=top
            break

stack.pop()
        }
    }

  }
 
return res.slice(0,n/2);
