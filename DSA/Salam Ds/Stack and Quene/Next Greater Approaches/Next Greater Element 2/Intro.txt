var nextGreaterElements = function(arr) {
   arr=arr.concat(...arr)
  let n=arr.length
  let res=Array(n).fill(-1), stack = [];

res[n-1]=-1;

for (let i=arr.length-2;i>=0;--i){
while(stack.length){
    let top =stack[stack.length-1]
    if (arr[i]>=top){
    stack.pop()
    }else{
     res[i]=top
      break
    }
}
if (stack.length==0){
    res[i]=-1;
}
stack.push(arr[i])  

};
return res.slice(0,n/2)
}
