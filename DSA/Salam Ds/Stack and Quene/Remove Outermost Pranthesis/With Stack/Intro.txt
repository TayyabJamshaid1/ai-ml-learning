var removeOuterParentheses = function(s) {
    let map={"(":")"},stack=[],str2="";

for (let i=0;i<s.length;i++){
   if (map[s[i]]){
       stack.push(s[i])
       if (stack.length>1){
        str2=str2+s[i]
    }

Recommended solution:
/**
 * @param {string} s
 * @return {string}
 */
var removeOuterParentheses = function(s) {
    let stack=[],str2="";

for (let i=0;i<s.length;i++){
   if (s[i]=="("){
       stack.push(s[i])
       if (stack.length>1){
        str2=str2+s[i]
    }
   }else{
    if (stack.length>1){
        str2=str2+s[i]
    }
    stack.pop();

   }
}
return str2;
};

Time complexity
O(n)

Space Complexity
O(n)
