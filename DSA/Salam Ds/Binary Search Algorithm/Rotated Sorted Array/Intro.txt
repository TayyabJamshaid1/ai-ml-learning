/**
 * @param {number[]} nums
 * @param {number} target
 * @return {number}
 */
var search = function(arr, target) {
    let l=0,r=arr.length-1;
    while(l<=r){
       let m=l+Math.floor(((r-l)/2))
      if (arr[m]===target){
        return m
      }
       if (arr[l]<=arr[m]){
        if (arr[m]>target && arr[l]<=target){
            r=m-1
        }else{
            l=m+1
        }
      }else if (arr[l]>arr[m]){
         if (arr[m]<target && arr[r]>=target){
            l=m+1 
        }else{
            r=m-1
        }
      }

    }
    return -1
};
