/**
 * @param {number} n
 * @return {number}
 */
var guessNumber = function(n) {
    let l=1;
    let r=n;
    while (l<=r){
        let m=Math.floor(l+((r-l)/2))
        let pick=guess(m)
        if (pick==0){
            return m;
        }else if(pick==1){
         l=m+1
        }else{
         r=m-1
        }
    }
    return -1
};
