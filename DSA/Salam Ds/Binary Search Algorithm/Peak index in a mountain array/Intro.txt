Now,Question requirement is we have to it of time complexity of O(logn)
So,below is the solution

var peakIndexInMountainArray = function(arr) {
let l=0,r=arr.length-1;
while(l<r){
    let m=l+Math.floor((r-l)/2)
    if (arr[m]<arr[m+1]){
        l=m+1
    }else{
         r=m
    }
}
return r

};
