It is an algorithm which searches for an element inside an array.Whether that element is avaiable in that array or not.If avaiable,what is the index of that matched element

We will iterate throughout an array.We will go to each element of an array to check whether this element is equal to target element or not.if we find target element,we just exit our loop and return the index of that position

My Solution:
let arr=[5,9,2,0,1],target=0,positionIndex=0;

for(let i=0;i<arr.length;i++){
  if (arr[i]==target){
    positionIndex=i
    break
  }
}
console.log(positionIndex)

Recommended and best solution:(I should follow this approach)
let arr=[5,9,2,0,1],target=0;
function linearSearch(arr,target){
for(let i=0;i<arr.length;i++){
  if (arr[i]==target){
   return i
  }
}
  return -1
}
let positionIndex=linearSearch(arr,target)
console.log(positionIndex)
