


function reshuffleOddEven(head){
if (!head || !head.next) return head;
let odd=head
let even=evenStart=head.next

while(odd.next && even.next){   //we use odd.next && even.next condition because loop k andr hum odd.next & even.next k behalf pr he values ly rhy ha
odd.next=odd.next.next      //see here and below we are using odd.next & even.next
even.next=even.next.next
odd=odd.next
even=even.next
}

odd.next=evenStart
return head
}

Time complexity is O(n),space complexity is O(1)
