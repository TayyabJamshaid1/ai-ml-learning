b)By Ds approach & Recommended solution:
let words=["leet","code"],finder="e",indexes=[]
for(let i=0;i<words.length;i++){
 for (let j=0;j<words[i].length;j++){
   if (words[i][j]==finder){
     indexes.push(i)
     break
   }
 }
}
console.log(indexes)
