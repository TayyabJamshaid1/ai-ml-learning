let arr = ["flower", "flight", "flow"], x = 0;

function largePossibleOddNumber(strs) {

    // The longest possible common prefix cannot be longer than the first string.
    while (x < strs[0].length) {

        // Take the character at the current position from the first string.
        // We will compare this character with the same position in every other string.
        let ch = strs[0][x];

        // Start from the second string because the character from the first
        // string is already stored in "ch".
        for (let i = 1; i < strs.length; i++) {

            // If any string has a different character at this position,
            // or this position does not exist in that string,means string has been ended, then the
            // common prefix ends just before this position.
            if (strs[i][x] !== ch || x == strs[i].length) {

                // Return everything before index x because index x is
                // the first position where the common prefix breaks.
                return strs[0].substring(0, x);
            }
        }

        // All strings had the same character at this position,
        // so move to the next character and continue checking.
        ++x;
    }

    // If we reach the end of the first string, every character of the
    // first string matched in all other strings, so the entire first
    // string is the common prefix.
    return strs[0];
}

let res = largePossibleOddNumber(arr);

console.log(res);

Initially,we assume that time complexity is O(n^2) because of nested loops but in reality it is not.
"O(n²) happens when two nested loops both run based on the same input size n.
If they depend on different input sizes, use different variables, e.g. O(m × n)."
Example of O(n^2):
et arr = [10, 20, 30, 40, 50];

You want to compare every element with every other element:

for (let i = 0; i < arr.length; i++) {

    for (let j = 0; j < arr.length; j++) {

        console.log(arr[i], arr[j]);

    }
}

Here both loops depend on the same array.

If there are 5 elements:

Outer loop → 5 times
Inner loop → 5 times for EACH outer iteration

Total = 5 × 5 = 25

If there are 100 elements:

100 × 100 = 10,000

If there are 1,000:

1,000 × 1,000 = 1,000,000

Therefore:

O(n × n) = O(n²)
So,If two nested loops both iterate based on the same input size n, then it is O(n²).
----------------------------------


In our example,There are 2 nested loops, but they iterate over different things:

while (x < strs[0].length)      // characters
    for (let i = 1; i < strs.length; i++)  // strings
Outer loop → iterates through characters → m
Inner loop → iterates through strings → n

Therefore:

Time Complexity = O(m × n)

Example:

5 strings × 10 characters
= 5 × 10
= 50 possible comparisons

So nested loops ≠ automatically O(n²).
First check what each loop is iterating over.
