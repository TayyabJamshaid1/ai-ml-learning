When one class(child/derived) derives properties & attributes of another class(parent/base)

Example:
class Car:
      def __init__(self,car_no,car_model):
            self.car_no=car_no
            self.car_model=car_model
   
      def start(self):
          print("Car Start")

      def stop(self):
          print("Car Stop")

class ToyotaCar(Car): # this is how you can extend the class
      def __init__(self, car_no, car_model):
           super().__init__(car_no, car_model)     
      def clickButton(self):
           self.start()
           print(self.car_model)  

a1=ToyotaCar(12345,"kesa")
a1.start()
a1.clickButton()
