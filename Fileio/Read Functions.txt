print(line1,type(line1))

print(line2,type(line2))
line3=f.readline() # it will read only third line and return string
print(line3,type(line3))

You can also use while loop in readline:

f=open("verify.txt")

line=f.readline()
while (line!=""):
      print(line)
      line=f.readline()

f.close()
