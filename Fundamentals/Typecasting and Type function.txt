# as we know int,string,none ,boolean are all data types,now if we have to check the type of variables ,we can do it by type function

a = 31
check=type(a)
print(check)  #output: <class 'int'>

a = 31.4
check=type(a)
print(check)  #output: <class 'float'>

a = "Hello"
check=type(a)
print(check)  #output: <class 'str'>

---------------------------
we can also change the types
d="31.4"
f=float(d)
print(f)  #output: 31.4

d="31"
f=int(d)
print(f)  #output: 31

d=78
j=str(d)
print(j)  #output: 78 as string
